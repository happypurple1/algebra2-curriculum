% --- EXAMPLE 1 ---
\textbf{Example 1:} \quad $(x^3 + 5x^2 + 8x + 4) \div (x + 1)$

\vspace{0.2cm}
\begin{center}
\renewcommand{\arraystretch}{2.5}
\large
\begin{tabular}{|c||p{0.8in}|p{0.8in}|p{0.8in}|}
\hline
$\times$ & \ans{$x^2$} & \ans{$4x$} & \ans{$4$} \\
\hline\hline
$x$ & \ans{$x^3$} & \ans{$4x^2$} & \ans{$4x$} \\
\hline
$+1$ & \ans{$x^2$} & \ans{$4x$} & \ans{$4$} \\
\hline
\end{tabular}
\end{center}

\vspace{0.1cm}
\textbf{Quotient:} \ans{$x^2 + 4x + 4$} \hfill \textbf{Remainder:} \ans{0}

\vspace{0.25cm}
\hrule
\vspace{0.3cm}

% --- EXAMPLE 2 ---
\textbf{Example 2:} \quad $(2x^4 + 3x^2 - 10) \div (x + 2)$ \hfill \textit{*Missing terms!}

\vspace{0.15cm}
Rewrite with placeholders: \ans{$2x^4 + 0x^3 + 3x^2 + 0x - 10$}

\vspace{0.2cm}
\begin{center}
\renewcommand{\arraystretch}{2.3}
\large
\begin{tabular}{|c||p{0.6in}|p{0.6in}|p{0.6in}|p{0.6in}|}
\hline
$\times$ & \ans{$2x^3$} & \ans{$-4x^2$} & \ans{$11x$} & \ans{$-22$} \\
\hline\hline
$x$ & \ans{$2x^4$} & \ans{$-4x^3$} & \ans{$11x^2$} & \ans{$-22x$} \\
\hline
$+2$ & \ans{$4x^3$} & \ans{$-8x^2$} & \ans{$22x$} & \ans{$-44$} \\
\hline
\end{tabular}
\end{center}

\vspace{0.1cm}
\textbf{Quotient:} \ans{$2x^3 - 4x^2 + 11x - 22$} \hfill \textbf{Remainder:} \ans{34}

{\small \ans{Note: Box gives $-44$ but we need $-10$, so remainder $= -10 - (-44) = 34$}}

\vspace{0.2cm}
\hrule
\vspace{0.3cm}

% --- YOU TRY ---
\textbf{You Try:} \quad $(3x^3 + 10x^2 + x - 6) \div (x + 2)$

\vspace{0.2cm}
\begin{center}
\renewcommand{\arraystretch}{2.5}
\large
\begin{tabular}{|c||p{0.8in}|p{0.8in}|p{0.8in}|}
\hline
$\times$ & \ans{$3x^2$} & \ans{$4x$} & \ans{$-7$} \\
\hline\hline
$x$ & \ans{$3x^3$} & \ans{$4x^2$} & \ans{$-7x$} \\
\hline
$+2$ & \ans{$6x^2$} & \ans{$8x$} & \ans{$-14$} \\
\hline
\end{tabular}
\end{center}

\vspace{0.1cm}
\textbf{Quotient:} \ans{$3x^2 + 4x - 7$} \hfill \textbf{Remainder:} \ans{8}

{\small \ans{Note: Box gives $-14$ but we need $-6$, so remainder $= -6 - (-14) = 8$}}
